# Динамическое окружение
Когда мы говорим об окружении, мы говорим о виртуальном выполнении кода программы.
*Динамическое окружение* - это совокупность значений связанных имён в момент выполнения конкретной инструкции программного кода

Для простоты на данный момент будем представлять динамическое окружение как список связанных значений
```
[(имя, значение), ...]
```


```
[(a, 25), (b, 13), (c, 78), (a, 13)]
a + b - c
25 + 13 - 78
```

## Как формируется динамическое окружение в SML
```sml
val a = 13
val b = 11
val c = 8
val a = 25
val d = let val a = 2
			val b = 8
		in a + b - c
		end

fun f (a, x) = a + x - c
fun g (a, x) = if a > 0 then 2 + g (a - 2, x * 2) else 1
val e = f (a, b)
val h = g (7, 3)
val j = a + b - c
val g1 = g
val j1 = g2 (1, 3)
```

Тогда динамическое окружение будет именоваться следующим образом:
```
[] => [(a, 13)]
[(b, 11), (a, 13)]
[(c, 8), (b, 11), (a, 13)]
[(a, 25), (c, 8), (b, 11), (a, 13)]  # Эффект затенения для a
[*(a, 2), (a, 25), (c, 8), (b, 11), (a, 13)] # (a, 2) - временное
[*(b, 8), *(a, 2), (a, 25), (c, 8), (b, 11), (a, 13)] # *(b, 8) - временное
2 + 8 - 8 => [(d, 2), (a, 25), (c, 8), (b, 11), (a, 13)]


[((f, [a, x], a + x - c), [(d, 2), (a, 25), (c, 8), (b, 11), (a, 13)]), (d, 2), (a, 25), (c, 8), (b, 11), (a, 13)]
[(g, (g, [a, x], if ...), [((f, [a, x], a + x - c), [(d, 2), (a, 25), (c, 8), (b, 11), (a, 13)]), (d, 2), (a, 25), (c, 8), (b, 11), (a, 13)], [((f, [a, x], a + x - c), [(d, 2), (a, 25), (c, 8), (b, 11), (a, 13)]), (d, 2), (a, 25), (c, 8), (b, 11), (a, 13)], (d, 2), (a, 25), (c, 8), (b, 11), (a, 13)]), (d, 2), (a, 25), (c, 8), (b, 11), (a, 13)]
f (25, 11) => [(f, ((f, [a, x], a + x - c), [(d, 2), (a, 25), (c, 8), (b, 11), (a, 13)]), (d, 2), (a, 25), (c, 8), (b, 11), (a, 13)] => [(a, 25), (x, 11), (f, ((f, [a, x], a + x - c), [(d, 2), (a, 25), (c, 8), (b, 11), (a, 13)]), (d, 2), (a, 25), (c, 8), (b, 11), (a, 13)] => a + x - c = 25 + 11 - 8 = 28 => [(e, 25), (g, (g, [a, x], if ...), [((f, [a, x], a + x - c), [(d, 2), (a, 25), (c, 8), (b, 11), (a, 13)]), (d, 2), (a, 25), (c, 8), (b, 11), (a, 13)], [((f, [a, x], a + x - c), [(d, 2), (a, 25), (c, 8), (b, 11), (a, 13)]), (d, 2), (a, 25), (c, 8), (b, 11), (a, 13)], (d, 2), (a, 25), (c, 8), (b, 11), (a, 13)]), (d, 2), (a, 25), (c, 8), (b, 11), (a, 13)]
[(g, ((g ...), [])), (x, 3), (a, 7), (f, ..), (d, 2), (a, 25), (c, 8), (b, 11), (a, 13)] => [(g, ((), [])), (x, 6), (a, 5), (f, ...), (d,2), (a, 25), (c, 8), (b, 11), (a, 13)] => [(g, ((), [])), (x,12), (a, 3),...] => [(h, 9), (e, 25), (g, (g, [a, x], if ...), [((f, [a, x], a + x - c), [(d, 2), (a, 25), (c, 8), (b, 11), (a, 13)]), (d, 2), (a, 25), (c, 8), (b, 11), (a, 13)], [((f, [a, x], a + x - c), [(d, 2), (a, 25), (c, 8), (b, 11), (a, 13)]), (d, 2), (a, 25), (c, 8), (b, 11), (a, 13)], (d, 2), (a, 25), (c, 8), (b, 11), (a, 13)]), (d, 2), (a, 25), (c, 8), (b, 11), (a, 13)]
[(e, 28), (g, (g, [a, x], if ...), [((f, [a, x], a + x - c), [(d, 2), (a, 25), (c, 8), (b, 11), (a, 13)]), (d, 2), (a, 25), (c, 8), (b, 11), (a, 13)], [((f, [a, x], a + x - c), [(d, 2), (a, 25), (c, 8), (b, 11), (a, 13)]), (d, 2), (a, 25), (c, 8), (b, 11), (a, 13)], (d, 2), (a, 25), (c, 8), (b, 11), (a, 13)]), (d, 2), (a, 25), (c, 8), (b, 11), (a, 13)]
```

Параметры, которые используются при описании функции называются **формальными параметрами**

Параметры, которые используются при вызове функции называются **фактическими параметрами**

**Замыкание функции** - совокупность определения функции вместе с динамическим окружением в котором эта функция была определена
```
(f, [a, x], a + x - c) # пример замыкания
```

# Статическое окружение
Статическое окружение не меняется во время выполнения программы

Определяется до выполнения программы, содержит информацию, неизменную во время выполнения программы

```sml
val a = 13
val b = 11
val c = 8
val a = 25
val d = let val a = 2
			val b = 8
		in a + b - c
		end

fun f (a, x) = a + x - c
fun g (a, x) = if a > 0 then 2 + g (a - 2, x * 2) else 1
val e = f (a, b)
val h = g (7, 3)
val j = a + b - c
val g1 = g
```

```
[] => [(a, int)]
[(b, int), (a, int)]
[(c, int), (b, int), (a, int)]
[(a, int), (c, int), (b, int), (a, int)]
[(d, int), (c, int), (b, int), (a, int)]




[(f, int * int -> int), (d, int), (c, int), (b, int), (a, int)]
[(g, int * int -> int), (f, int * int -> int), (d, int), (c, int), (b, int), (a, int)]
[(e, int), (g, int * int -> int), (f, int * int -> int), (d, int), (c, int), (b, int), (a, int)]
[(h, int), (e, int), (g, int * int -> int), (f, int * int -> int), (d, int), (c, int), (b, int), (a, int)]
[(j, int), (h, int), (e, int), (g, int * int -> int), (f, int * int -> int), (d, int), (c, int), (b, int), (a, int)]
[(g1, int * int -> int), (j, int), (h, int), (e, int), (g, int * int -> int), (f, int * int -> int), (d, int), (c, int), (b, int), (a, int)]
```

Статическое окружение необходимо только при предварительном анализе программы, при выполнении программы статическое окружение не нужно, поэтому после формирования статического окружения оно отбрасывается, в дальнейшем мы его не рассматриваем

# Модели вычислений

```sml
fun square x = x * x
fun sumOfSquares (x, y) = square x + square y

sumOfSquares(5 - 3, 2 + 1)
```

## Аппликативная модель вычислений
При вызове функции сначала вычисляются параметры функции
```
sumOfSquares(2, 3)
square 2 + square 3
2 * 2 + 3 * 3 => 4 + 9 = 13
```
