$$
\begin{gathered}
u = xy + \frac{y}{x} \\
\frac{\partial u}{\partial x} = y - \frac{y}{x^2} \\
\frac{\partial u}{\partial y} = x + \frac{1}{x} \\
\frac{\partial^2 u}{\partial x^2} = -\frac{2}{x^3}y \\
\frac{\partial^2 u}{\partial x \partial y} = 1 - \frac{1}{x^2} \\
\frac{\partial^2 u}{\partial y^2} = 0 \\
\frac{\partial^2 u}{\partial y \partial x} = 1 - \frac{1}{x^2}
\end{gathered}
