1. Общий признак
$$
\text{Если } \sum x_{k} \text{ сходится} \implies \lim_{ k \to \infty } x_{k} = 0, \text{ ряд сходится}
$$
2. 
$$
\begin{gathered}
\text{Если } 0 \leq x_{k} \leq y_{k} \text{ и } \sum y_{k} \text{ сходится } \implies \sum y_{k} \text{ расходится } \\
\text{Если же } \sum x_{k} \text{ расходится } \implies \sum y_{k} \text{ расходится}
\end{gathered}
$$

3. Признак Коши
$$
\begin{gathered}
\alpha = \lim_{ k \to \infty } \sqrt[k]{ | x_{k}| } \\
\begin{array}l
\alpha < 1  &  & \sum x_{k} \text{ сходится} \\
\alpha > 1  &  & \sum x_{k} \text{ расходится}
\end{array}
\end{gathered}
$$
4. Признак Даламебра
$$
\begin{gathered}
p = \overline{\lim_{ l \to \infty } } | \frac{x_{k+1}}{x_{l}}|, \text{ если } \beta < 1 \implies \sum_{k=1}^\infty x_{l}  \text{ сходится}  \\
\text{Если } | \frac{x_{k+1}}{x_{k}}| \geq 1 \qquad k \geq k_{0} \qquad \sum x_{k} \text{ расходится}
\end{gathered}
$$
5. Признак Раабе
$$
\begin{gathered}
\lim_{ n \to \infty } n \left( \frac{x_{n}}{x_{n+1}} - 1 \right) = p \\
p > 1 \implies \sum x_{n} \text{ сходится} \\
p < 1 \implies \sum x_{n} \text{ расходится}
\end{gathered}
$$

6. Пределный признак
$$
\begin{gathered}
\text{Если } \exists \lim_{ k \to \infty } \frac{x_{k}}{y_{k}} = c \neq 0, \infty \implies \sum x_{k} \text{ и } \sum y_{k} \text{ ведут себя одинаково в смысле сходимости}
\end{gathered}
$$

7. Интегральный признак Коши
$$
\begin{gathered}
\text{Если } f(x) \geq \text{ непрерывная, невозрастающая } \int_{-\infty}^{+\infty} f(x) dx \text{ и } \sum_{k=1}^\infty f(x) \text{ ведут себя } \\
\text{одинаково в смысле сходимости}
\end{gathered}
$$
