# Метод двух указателей
## Окна фиксированного размера
Для всех элементов массива размера k рассмотрим отрезки длины k:
```
0: k
1: k + 1
2: k + 2
3: k + 3
```

Пусть f - время обработки одного элемента
Время для отрезка длины k = $O(k\cdot f)$

1. Вычислить функцию на отрезке $[0, k)$. Работает за $O(k\cdot f)$
2. Двигаем границы:
   - правую на один вправо
   - левую на 1 вправо
    При движении границы добавляем или удаляем из рассмотрения элементы. Сохраняем длину, но отрезок сдвинулся на позицию вправо за итерацию. Ещё будет n-k отрезков, каждый из них обрабатывается за два элемента и, соответсвенно, нам потребуется на обработку $O((n-k)\cdot f)\implies O(n\cdot f)$

---
Как это сделать? 

```python
sums = []
for i in range (n - k + 1):
	cur = 0
```

Если мы добавим элемент в `distinctOnLenK`, то у него было 0 вхождений, это новый элемент

```cpp
#include <ctime>
#include <iostream>
#include <map>
#include <vector>

using namespace std;

vector<long long> sumOnLenK(const vector<int> &a, int k) {
  // сумма длины на всех подотрезках длины k
  int n = a.size();
  long long s = 0;
  for (int i = 0; i < k; i++) {
    s += a[i];
  }

  vector<long long> res = {s};
  int l = 0;
  int r = k;

  while (r < n) {
    s += a[r];
    r++;
    s -= a[l];
    l++;
    res.push_back(s);
  }

  return res;
}

int addElement(map<int, int> &d, int x) {
  // добавить в словарь элемент x
  // возвращает 1, если новый элемент или 0, если старый
  d[x]++;
  if (d[x] == 1)
    return 1;
  else
    return 0;
}

int delElement(map<int, int> &d, int x) {
  // удалить из словаря элемент x
  // возвращает -1, если это единственный x или 0, если есть другие вхождения
  d[x]--;
  if (d[x] == 0)
    return -1;
  else
    return 0;
}

vector<int> distinctOnLenK(const vector<int> &a, int k) {
  int n = a.size();
  map<int, int> d; // поддерживает количество различных на текущем отрезке через
                   // количество вхождений каждого элемента
  int s = 0;
  for (int i = 0; i < k; i++)
    s += addElement(d, a[i]);

  vector<int> res = {s};
  int l = 0;
  int r = k;
  while (r < n) {
    s += addElement(d, a[r]);
    r++;
    s -= delElement(d, a[l]);
    l++;
    res.push_back(s);
  }
  return res;
}

int main() { return 0; }

```

## Самый короткий/самый длинный "хороший" отрезок
*Пример задачи*: 
Самый короткий отрезок с хотя бы k различными элементами
Самый короткий отрезок: 
- Если отрезок "плохой", то любой его подотрезок тоже плохой. Это условие очень важно, без него данный метод решения не работает

Идея метода:
Начнём с пустого отрезка
`l = 0, r = 0`
Движение правой границы "улучшает" отрезок
Движение левой границы "ухудшает" отрезок

Сделаем такой цикл, в котором будем действовать следующим образом: 
> Пока отрезок "плохой" => двигаем правую границу
> Если отрезок "хороший" => обновим ответ, двигаем левую границу

```cpp
#include <ctime>
#include <iostream>
#include <map>
#include <vector>

using namespace std;

const int INF = 1e9;

int addElement(map<int, int> &d, int x) {
  // добавить в словарь элемент x
  // возвращает 1, если новый элемент или 0, если старый
  d[x]++;
  if (d[x] == 1)
    return 1;
  else
    return 0;
}

int delElement(map<int, int> &d, int x) {
  // удалить из словаря элемент x
  // возвращает -1, если это единственный x или 0, если есть другие вхождения
  d[x]--;
  if (d[x] == 0)
    return -1;
  else
    return 0;
}

int shortestWithKDistinct(const vector<int> &a, int k) {
  // самый короткий, в котором k различных элементов
  // не работает, если пустой подотрезок хороший
  int n = a.size();
  int ans = INF;
  map<int, int> d;
  int s = 0;

  int l = 0, r = 0;

  while (l < n) {
    while (r < n && s < k) {
      s += addElement(d, a[r]);
      r++;
    }
    if (s >= k) {
      ans = min(ans, r - l);
    }
    s += delElement(d, a[l]);
    l++;
  }

  return ans;
}

```
