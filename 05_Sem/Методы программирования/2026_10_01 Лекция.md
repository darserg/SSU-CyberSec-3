ПонятЬ, как применить бинарный поиск по ответу $\implies$ понять, как ввести функцию $f(x)$ - верно ли, что ответ $\geq x$ или $>x$?

## Задача 1. Целочисленный бинарный поиск по ответу 
Есть $n$ предприятий, $i$-е предприятие выпускает 1 холодильник за $a_{i}$ дней, найти количество дней $t$, за которое суммарно выпущено $\geq k$ холодильников

```
n = 5
3 2 4 1 2
k = 15
```

```
t = 7
2 3 1 7 3

t = 6
2 3 1 6 3
```

Какую функцию можно объявить для использования БП по ответу? 

Попробуем для какого-то $x$ понять, верно ли, что за $x$ дней будет выпущено $< k$ холодильников

Определение функции точно совпадает с 
> верно ли, что ответ $>x$?

```cpp
#include <iostream>
#include <vector>

using namespace std;

bool check_fridges(const vector<int> &a, long long k, long long x)
{
  // верно ли, что ответ > x?
  // верно ли, что за x дней выпущено < k холодильников?

  long long s = 0;
  for (auto p : a)
  {
    s += (x / p);
    if (s >= k) // проверка помогает избежать переполнения и в некоторых случаях может ускорить высисления
      return false;
  }

  return true;
}

long long get_days(const vector<int> &a, long long k)
{
  long long l = 0;
  long long r = 1;

  // нужно найти первый момент, когда функция вернёт false
  while (check_fridges(a, k, r))
    r *= 2;

  long long ans = r;

  while (l <= r)
  {
    long long m = l + (r - l) / 2;
    if (check_fridges(a, k, m))
    {
      l = m + 1;
    }
    else
    {
      ans = m;
      r = m - 1;
    }
  }

  return ans;
}

int main()
{
  cout << get_days(vector<int>({3, 2, 4, 1, 2}), 15) << endl;
  cout << get_days(vector<int>({1, 1, 1}), 2) << endl;
}
```


```python
def check_fridges(a, k, x):
    # верно ли, что ответ > x?
    # верно ли, что за x дней выпущено < k холодильников?
    s = 0

    for p in a:
        s += (x // p)
        if s >= k:
            return False

    return True


def get_days(a, k):
    l = 0
    r = 1

    while check_fridges(a, k, r):
        r *= 2

    ans = r

    while l <= r:
        m = (l + r) // 2
        if check_fridges(a, k, m): 
            l = m + 1
        else:
            ans = m
            r = m - 1

    return ans


if __name__ == '__main__':
    print(get_days([3, 2, 4, 1, 2], 15))
    print(get_days([1, 1, 1], 2))

```

## Задача 3. Нецелочисленный БП по ответу
Есть $n$ растворов кислоты, $V_{i}$ - объём раствора, $c_{i}$ - концентрация раствора

Мы хотим получить кислоту с концентрацией $\geq p$. Какой максимальный объём можно получить?

$f(x)$ - верно ли, что можно получить раствор с концентрацией $\geq p$ объёма хотя бы x
> верно ли, что ответ >= x


```text title:"3 р-ра"
100 0.75
50 0.9
300 0.5
```
Нужно получить максимальное количество концентрации $\geq 0.6$


### f(x):
1. Отсортируем растворы по убыванию концентрации и получим первыми несколькими растворами объём $x$
2. Проверим, что полученная концентрация $\geq p$

```cpp title:"3 задача C++"
#include <iostream>
#include <vector>
#include <random>
#include <ctime>

using namespace std;

bool check_concentration(const vector<pair<double, double>> &a, double x, double p)
{
  // a - пары вида (концентрация, объём)
  // отсортированные по убывнию концентрации

  // верно ли, что раствор объёма x из наиболее концентрированных компонентов
  // имеет концентрацию хотя бы p

  double volume = 0.0; // объём раствора
  double acid = 0.0;   // объём кислоты

  for (int i = 0; i < a.size(); i++)
  {
    double add = min(x - volume, a[i].second);
    volume += add;
    acid += add * a[i].first;
  }

  return abs(volume - x) < 1e-9 && acid - p * volume > -1e-9;
}

double get_max_volume(const vector<double> &v, const vector<double> &c, double p)
{
  int n = v.size();
  vector<pair<double, double>> a(n);

  for (int i = 0; i < n; i++)
    a[i] = {c[i], v[i]};

  sort(a.begin(), a.end());
  reverse(a.begin(), a.end());

  double l = 0.0;
  double r = 0.0;

  for (int i = 0; i < n; i++)
    r += v[i];

  int iters = log2((r - l) / 1e-9) + 1;

  for (int i = 0; i < iters; i++)
  {
    double m = (l + r) / 2.0;
    if (check_concentration(a, m, p))
      l = m;
    else
      r = m;
  }

  return (l + r) / 2.0;
}

int main()
{
  cout << get_max_volume(vector<double>({100, 50, 300}, {0.75, 0.9, 0.5}), 0.6)
}
```

from math import log


def check_concentration(a, x, p):
    volume = 0.0
    acid = 0.0

    for i in range(len(a)):
        add = min(x - volume, a[i][1])
        volume += add
        acid += add * a[i][0]

    return abs(volume - x) < 1e-9 and acid - p * volume > -1e-9


def get_max_value(v, c, p):
    n = len(v)
    a = [[c[i], v[i]] for i in range(n)]
    a = sorted(a)
    a = a[::-1]

    l = 0.0
    r = sum(v)

    iters = int(log((r-l) / 1e-9, 2)) + 1
    for i in range(iters):
        m = (l + r) / 2
        if check_concentration(a, m, p): 
            l = m
        else:
            r = m
    return (l + r) / 2.0
